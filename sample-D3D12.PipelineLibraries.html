// =====================================================================
// D3D12.PipelineLibraries - cache graphics PSOs with ID3D12PipelineLibrary.
//
// Pipeline State Objects are deliberately immutable: the driver can compile a
// complete blend/raster/depth/input-layout/shader combination into the exact GPU
// program it wants to execute. That is great while drawing, but cold PSO creation
// can hitch if an engine discovers many material permutations at runtime.
//
// This sample draws the familiar rotating cube with two PSOs (solid and
// wireframe), but acquires them through an ID3D12PipelineLibrary. The first run
// compiles normally, stores each PSO in the library, serializes the library to
// CubeD3D12.PipelineLibraries.cache, and logs timings. Later runs try to load
// those named PSOs directly from the on-disk blob.
//
// [LEARN] Pipeline libraries solve the PSO explosion problem by turning many
// cold CreateGraphicsPipelineState calls into cheap named lookups. Cache blobs
// are adapter/driver specific, so stale blobs can fail after a driver update;
// engines must log the miss, discard the blob, and rebuild gracefully.
// ===================================================================

static void logf_line(const char *fmt, ...)
{
    va_list ap; FILE *f = fopen("CubeD3D12.PipelineLibraries.log", "a");
    if (!f) return;
    va_start(ap, fmt); vfprintf(f, fmt, ap); va_end(ap);
    fputc('\n', f); fclose(f);
}
static void log_blob(const char *label, ID3DBlob *b)
{
    if (b) logf_line("%s: %.*s", label, (int)ID3D10Blob_GetBufferSize(b),
        (const char *)ID3D10Blob_GetBufferPointer(b));
}

static Mat4 mat_mul(Mat4 a, Mat4 b)
{
    Mat4 r; int i, j, k;
    for (i = 0; i < 4; i++)
        for (j = 0; j < 4; j++) {
            float s = 0.0f;
            for (k = 0; k < 4; k++) s += a.m[i * 4 + k] * b.m[k * 4 + j];
            r.m[i * 4 + j] = s;
        }
    return r;
}
static Mat4 mat_rot_x(float a) { float c=cosf(a),s=sinf(a); Mat4 r={{1,0,0,0, 0,c,s,0, 0,-s,c,0, 0,0,0,1}}; return r; }
static Mat4 mat_rot_y(float a) { float c=cosf(a),s=sinf(a); Mat4 r={{c,0,-s,0, 0,1,0,0, s,0,c,0, 0,0,0,1}}; return r; }
static Mat4 mat_lookat_lh(float ex, float ey, float ez)
{
    float zx=-ex,zy=-ey,zz=-ez;
    float zl=sqrtf(zx*zx+zy*zy+zz*zz); zx/=zl; zy/=zl; zz/=zl;
    { float xx=1*zz-0*zy, xy=0*zx-0*zz, xz=0*zy-1*zx;
      float xl=sqrtf(xx*xx+xy*xy+xz*xz); xx/=xl; xy/=xl; xz/=xl;
      { float yx=zy*xz-zz*xy, yy=zz*xx-zx*xz, yz=zx*xy-zy*xx;
        float dx=-(xx*ex+xy*ey+xz*ez), dy=-(yx*ex+yy*ey+yz*ez), dz=-(zx*ex+zy*ey+zz*ez);
        Mat4 r={{ xx,yx,zx,0, xy,yy,zy,0, xz,yz,zz,0, dx,dy,dz,1 }}; return r; } }
}
static Mat4 mat_perspective_lh(float fovY, float aspect, float zn, float zf)
{
    float ys=1.0f/tanf(fovY*0.5f), xs=ys/aspect;
    Mat4 r={{ xs,0,0,0, 0,ys,0,0, 0,0,zf/(zf-zn),1, 0,0,-zn*zf/(zf-zn),0 }}; return r;
}

static void signal_and_wait(void)
{
    HRESULT hr = ID3D12CommandQueue_Signal(command_queue, queue_fence, ++fence_value);
    assert(SUCCEEDED(hr)); (void)hr;
    while (ID3D12Fence_GetCompletedValue(queue_fence) != fence_value)
        SwitchToThread();
}

static void execute_commands(void)
{
    ID3D12CommandList *lists[1];
    ID3D12GraphicsCommandList_Close(command_list);
    lists[0] = (ID3D12CommandList *)command_list;
    ID3D12CommandQueue_ExecuteCommandLists(command_queue, 1, lists);
    signal_and_wait();
    ID3D12CommandAllocator_Reset(command_allocator);
    ID3D12GraphicsCommandList_Reset(command_list, command_allocator, NULL);
}

static ID3D12Resource *create_upload_buffer(const void *data, UINT size)
{
    D3D12_HEAP_PROPERTIES hp; D3D12_RESOURCE_DESC rd;
    ID3D12Resource *res = NULL; void *mapped = NULL; D3D12_RANGE none = { 0, 0 };
    memset(&hp, 0, sizeof(hp)); hp.Type = D3D12_HEAP_TYPE_UPLOAD;
    memset(&rd, 0, sizeof(rd));
    rd.Dimension = D3D12_RESOURCE_DIMENSION_BUFFER;
    rd.Width = size; rd.Height = 1; rd.DepthOrArraySize = 1; rd.MipLevels = 1;
    rd.Format = DXGI_FORMAT_UNKNOWN; rd.SampleDesc.Count = 1;
    rd.Layout = D3D12_TEXTURE_LAYOUT_ROW_MAJOR;
    ID3D12Device_CreateCommittedResource(device, &hp, D3D12_HEAP_FLAG_NONE, &rd,
        D3D12_RESOURCE_STATE_GENERIC_READ, NULL, &IID_ID3D12Resource, (void **)&res);
    ID3D12Resource_Map(res, 0, &none, &mapped);
    memcpy(mapped, data, size);
    ID3D12Resource_Unmap(res, 0, NULL);
    return res;
}

static void create_depth_buffer(int width, int height)
{
    D3D12_HEAP_PROPERTIES hp; D3D12_RESOURCE_DESC rd; D3D12_CLEAR_VALUE cv;
    D3D12_CPU_DESCRIPTOR_HANDLE dsv;
    memset(&hp, 0, sizeof(hp)); hp.Type = D3D12_HEAP_TYPE_DEFAULT;
    memset(&rd, 0, sizeof(rd));
    rd.Dimension = D3D12_RESOURCE_DIMENSION_TEXTURE2D;
    rd.Width = (UINT)width; rd.Height = (UINT)height;
    rd.DepthOrArraySize = 1; rd.MipLevels = 1;
    rd.Format = DXGI_FORMAT_D32_FLOAT; rd.SampleDesc.Count = 1;
    rd.Flags = D3D12_RESOURCE_FLAG_ALLOW_DEPTH_STENCIL;
    memset(&cv, 0, sizeof(cv)); cv.Format = DXGI_FORMAT_D32_FLOAT; cv.DepthStencil.Depth = 1.0f;
    ID3D12Device_CreateCommittedResource(device, &hp, D3D12_HEAP_FLAG_NONE, &rd,
        D3D12_RESOURCE_STATE_DEPTH_WRITE, &cv, &IID_ID3D12Resource, (void **)&depth_buffer);
    ID3D12DescriptorHeap_GetCPUDescriptorHandleForHeapStart(dsv_heap, &dsv);
    ID3D12Device_CreateDepthStencilView(device, depth_buffer, NULL, dsv);
}

static void get_swap_chain_buffers(void)
{
    D3D12_CPU_DESCRIPTOR_HANDLE h;
    ID3D12DescriptorHeap_GetCPUDescriptorHandleForHeapStart(rtv_heap, &h);
    IDXGISwapChain1_GetBuffer(swap_chain, 0, &IID_ID3D12Resource, (void **)&rtv_buffers[0]);
    IDXGISwapChain1_GetBuffer(swap_chain, 1, &IID_ID3D12Resource, (void **)&rtv_buffers[1]);
    ID3D12Device_CreateRenderTargetView(device, rtv_buffers[0], NULL, h);
    rtv_handles[0] = h;
    h.ptr += rtv_increment;
    ID3D12Device_CreateRenderTargetView(device, rtv_buffers[1], NULL, h);
    rtv_handles[1] = h;
}

static void read_cache_blob(void **blob, SIZE_T *size)
{
    FILE *f = fopen(CACHE_FILE, "rb"); long len;
    *blob = NULL; *size = 0; g_cache_size = 0;
    if (!f) { strcpy(g_cache_note, "cache file missing; cold run"); logf_line("cache miss: file missing"); return; }
    fseek(f, 0, SEEK_END); len = ftell(f); fseek(f, 0, SEEK_SET);
    if (len <= 0) { fclose(f); strcpy(g_cache_note, "cache file empty; cold run"); logf_line("cache miss: empty file"); return; }
    *blob = malloc((SIZE_T)len);
    if (!*blob) { fclose(f); strcpy(g_cache_note, "out of memory reading cache"); return; }
    if (fread(*blob, 1, (SIZE_T)len, f) != (SIZE_T)len) { free(*blob); *blob = NULL; fclose(f); strcpy(g_cache_note, "cache read failed"); return; }
    fclose(f); *size = (SIZE_T)len; g_cache_size = (SIZE_T)len;
    sprintf(g_cache_note, "read %llu bytes from cache", (unsigned long long)*size);
    logf_line("cache file read: %llu bytes", (unsigned long long)*size);
}

static void init_pipeline_library(void)
{
    void *blob = NULL; SIZE_T blob_size = 0; HRESULT hr;
    hr = ID3D12Device_QueryInterface(device, &IID_ID3D12Device1, (void **)&device1);
    if (FAILED(hr)) { logf_line("ID3D12Device1 unavailable hr=0x%08X", (unsigned)hr); strcpy(g_cache_note, "ID3D12Device1 unavailable"); return; }
    g_library_supported = 1;
    read_cache_blob(&blob, &blob_size);
    // [LEARN] CreatePipelineLibrary validates the blob against this adapter and
    // driver. Stale blobs fail cleanly, so the fallback is an empty library.
    hr = ID3D12Device1_CreatePipelineLibrary(device1, blob, blob_size, &IID_ID3D12PipelineLibrary, (void **)&pipeline_library);
    if (blob) free(blob);
    if (FAILED(hr)) {
        logf_line("CreatePipelineLibrary(blob) hr=0x%08X; creating empty library", (unsigned)hr);
        sprintf(g_cache_note, "stale/mismatched cache hr=0x%08X; rebuilt empty", (unsigned)hr);
        hr = ID3D12Device1_CreatePipelineLibrary(device1, NULL, 0, &IID_ID3D12PipelineLibrary, (void **)&pipeline_library);
    } else g_cache_loaded = (blob_size > 0);
    if (FAILED(hr)) { logf_line("CreatePipelineLibrary(empty) hr=0x%08X", (unsigned)hr); pipeline_library = NULL; }
}

static ID3D12PipelineState *create_or_load_pso(D3D12_GRAPHICS_PIPELINE_STATE_DESC *pd, int index)
{
    ID3D12PipelineState *pso = NULL;
    LARGE_INTEGER f, a, b; HRESULT hr = E_FAIL;
    QueryPerformanceFrequency(&f); QueryPerformanceCounter(&a);
    if (pipeline_library) {
        // [LEARN] Try a named load first. A hit skips shader/driver PSO work.
        hr = ID3D12PipelineLibrary_LoadGraphicsPipeline(pipeline_library, g_pso_stats[index].name_w,
            pd, &IID_ID3D12PipelineState, (void **)&pso);
        if (SUCCEEDED(hr)) g_pso_stats[index].hit = 1;
    }
    if (!pso) {
        // [LEARN] Miss path: compile the real PSO once, then store it by name.
        hr = ID3D12Device_CreateGraphicsPipelineState(device, pd, &IID_ID3D12PipelineState, (void **)&pso);
        if (SUCCEEDED(hr) && pso && pipeline_library) {
            HRESULT shr = ID3D12PipelineLibrary_StorePipeline(pipeline_library, g_pso_stats[index].name_w, pso);
            if (FAILED(shr)) logf_line("StorePipeline(%s) hr=0x%08X", g_pso_stats[index].name, (unsigned)shr);
        }
    }
    QueryPerformanceCounter(&b);
    g_pso_stats[index].hr = hr;
    g_pso_stats[index].ms = (double)(b.QuadPart - a.QuadPart) * 1000.0 / (double)f.QuadPart;
    logf_line("PSO %s %s hr=0x%08X time=%.3f ms", g_pso_stats[index].name,
        g_pso_stats[index].hit ? "HIT" : "MISS", (unsigned)hr, g_pso_stats[index].ms);
    return pso;
}

static void save_pipeline_library(void)
{
    SIZE_T size; void *blob; FILE *f; HRESULT hr;
    if (!pipeline_library) return;
    size = ID3D12PipelineLibrary_GetSerializedSize(pipeline_library);
    if (!size) { logf_line("pipeline library serialized size is zero"); return; }
    blob = malloc(size);
    if (!blob) { logf_line("malloc failed while saving pipeline library"); return; }
    hr = ID3D12PipelineLibrary_Serialize(pipeline_library, blob, size);
    if (SUCCEEDED(hr)) {
        f = fopen(CACHE_FILE, "wb");
        if (f) {
            fwrite(blob, 1, size, f); fclose(f); g_cache_saved = 1; g_cache_size = size;
            logf_line("serialized pipeline library: %llu bytes", (unsigned long long)size);
        } else logf_line("failed to open cache for write");
    } else logf_line("Serialize pipeline library hr=0x%08X", (unsigned)hr);
    free(blob);
}

static ID3D12PipelineState *build_pso(ID3DBlob *vs, ID3DBlob *ps, D3D12_FILL_MODE fill, int index)
{
    D3D12_INPUT_ELEMENT_DESC il[2]; D3D12_GRAPHICS_PIPELINE_STATE_DESC pd; UINT i;
    il[0].SemanticName = "POSITION"; il[0].SemanticIndex = 0;
    il[0].Format = DXGI_FORMAT_R32G32B32_FLOAT; il[0].InputSlot = 0; il[0].AlignedByteOffset = 0;
    il[0].InputSlotClass = D3D12_INPUT_CLASSIFICATION_PER_VERTEX_DATA; il[0].InstanceDataStepRate = 0;
    il[1].SemanticName = "COLOR"; il[1].SemanticIndex = 0;
    il[1].Format = DXGI_FORMAT_R32G32B32A32_FLOAT; il[1].InputSlot = 0; il[1].AlignedByteOffset = 12;
    il[1].InputSlotClass = D3D12_INPUT_CLASSIFICATION_PER_VERTEX_DATA; il[1].InstanceDataStepRate = 0;
    memset(&pd, 0, sizeof(pd));
    pd.pRootSignature = root_sig;
    pd.VS.pShaderBytecode = ID3D10Blob_GetBufferPointer(vs); pd.VS.BytecodeLength = ID3D10Blob_GetBufferSize(vs);
    pd.PS.pShaderBytecode = ID3D10Blob_GetBufferPointer(ps); pd.PS.BytecodeLength = ID3D10Blob_GetBufferSize(ps);
    pd.InputLayout.pInputElementDescs = il; pd.InputLayout.NumElements = 2;
    pd.PrimitiveTopologyType = D3D12_PRIMITIVE_TOPOLOGY_TYPE_TRIANGLE;

static int create_cube_pipeline(void)
{
    D3D12_ROOT_PARAMETER rp; D3D12_ROOT_SIGNATURE_DESC rsd;
    ID3DBlob *sig = NULL, *sigerr = NULL, *vs = NULL, *ps = NULL, *err = NULL; HRESULT hr;
    memset(&rp, 0, sizeof(rp));
    rp.ParameterType = D3D12_ROOT_PARAMETER_TYPE_32BIT_CONSTANTS; rp.Constants.ShaderRegister = 0;
    rp.Constants.Num32BitValues = 16; rp.ShaderVisibility = D3D12_SHADER_VISIBILITY_VERTEX;
    memset(&rsd, 0, sizeof(rsd)); rsd.NumParameters = 1; rsd.pParameters = &rp;
    rsd.Flags = D3D12_ROOT_SIGNATURE_FLAG_ALLOW_INPUT_ASSEMBLER_INPUT_LAYOUT;

    ID3D10Blob_Release(sig);
    hr = D3DCompile(g_shader_src, strlen(g_shader_src), NULL, NULL, NULL, "VSMain", "vs_5_0", 0, 0, &vs, &err);
    if (FAILED(hr)) { logf_line("VS hr=0x%08X", (unsigned)hr); log_blob("VS", err); if (err) ID3D10Blob_Release(err); return 0; }
    hr = D3DCompile(g_shader_src, strlen(g_shader_src), NULL, NULL, NULL, "PSMain", "ps_5_0", 0, 0, &ps, &err);
    if (FAILED(hr)) { logf_line("PS hr=0x%08X", (unsigned)hr); log_blob("PS", err); if (err) ID3D10Blob_Release(err); return 0; }
    pso_solid = build_pso(vs, ps, D3D12_FILL_MODE_SOLID, 0);
    pso_wire = build_pso(vs, ps, D3D12_FILL_MODE_WIREFRAME, 1);
    ID3D10Blob_Release(vs); ID3D10Blob_Release(ps);
    vbuffer = create_upload_buffer(CUBE_VERTS, sizeof(CUBE_VERTS));
    vbv.BufferLocation = ID3D12Resource_GetGPUVirtualAddress(vbuffer); vbv.StrideInBytes = sizeof(Vertex); vbv.SizeInBytes = sizeof(CUBE_VERTS);

    ibv.BufferLocation = ID3D12Resource_GetGPUVirtualAddress(ibuffer); ibv.Format = DXGI_FORMAT_R16_UINT; ibv.SizeInBytes = sizeof(CUBE_IDX);
    return (root_sig && pso_solid && pso_wire && vbuffer && ibuffer);
}
